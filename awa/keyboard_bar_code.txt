void Launcher::render_keyboard_bar()
{
    // If an app is running, do not render keyboard bar
    if (_data.running_app_id >= 0) {
        return;
    }

    // Background - Cyberpunk Vertical Dock
    // int margin_x = 0; // Unused
    // int margin_y = 0; // Unused
    
    // Clear with transparent/black
    GetHAL().canvasKeyboardBar.fillScreen(TFT_BLACK);
    
    // Draw vertical glass panel
    // Gradient from dark purple to black
    for(int i=0; i<GetHAL().canvasKeyboardBar.width(); i++) {
        // uint8_t alpha = 100 + (i * 155 / GetHAL().canvasKeyboardBar.width()); // Unused
        // Simple vertical gradient rect
        GetHAL().canvasKeyboardBar.fillRect(i, 0, 1, GetHAL().canvasKeyboardBar.height(), 
            GetHAL().canvasKeyboardBar.color565(20 + (i * 40 / GetHAL().canvasKeyboardBar.width()), 0, 30));
    }
    
    // Neon Border on the right
    GetHAL().canvasKeyboardBar.drawFastVLine(GetHAL().canvasKeyboardBar.width()-1, 0, GetHAL().canvasKeyboardBar.height(), 0xF81F); // Magenta

    // Decorative "Circuit" lines
    GetHAL().canvasKeyboardBar.drawFastHLine(0, 10, 15, 0xF81F);
    GetHAL().canvasKeyboardBar.drawFastHLine(0, GetHAL().canvasKeyboardBar.height()-10, 20, 0x07E0); // Green
    
    // render state
    int x      = 7;
    int y      = 20;
    int width  = 22;
    int height = 17;
    int gap_y  = 6; // Increased gap

    // Helper to draw button background
    auto drawBtnBg = [&](int bx, int by, bool active, uint16_t color) {
        if (active) {
            // Glow effect
            GetHAL().canvasKeyboardBar.fillRoundRect(bx-2, by-2, width+4, height+4, 4, color);
            GetHAL().canvasKeyboardBar.drawRoundRect(bx-2, by-2, width+4, height+4, 4, TFT_WHITE);
        } else {
            // Dim background
            GetHAL().canvasKeyboardBar.drawRoundRect(bx-1, by-1, width+2, height+2, 4, GetHAL().canvasKeyboardBar.color565(60, 60, 60));
        }
    };

    drawBtnBg(x, y, _data.keyboard_state.caps_lock, TFT_RED);
    if (_data.keyboard_state.caps_lock) {
        GetHAL().canvasKeyboardBar.pushImage(x, y, width, height, image_data_Aa);
    } else {
        GetHAL().canvasKeyboardBar.pushImage(x, y, width, height, image_data_Aa0);
    }

    y = y + height + gap_y;
    drawBtnBg(x, y, _data.keyboard_state.fn, TFT_ORANGE);
    if (_data.keyboard_state.fn) {
        GetHAL().canvasKeyboardBar.pushImage(x, y, width, height, image_data_fn);
    } else {
        GetHAL().canvasKeyboardBar.pushImage(x, y, width, height, image_data_fn0);
    }

    y = y + height + gap_y;
    drawBtnBg(x, y, _data.keyboard_state.ctrl, TFT_CYAN);
    if (_data.keyboard_state.ctrl) {
        GetHAL().canvasKeyboardBar.pushImage(x, y, width, height, image_data_crtl);
    } else {
        GetHAL().canvasKeyboardBar.pushImage(x, y, width, height, image_data_ctrl0);
    }

    y = y + height + gap_y;
    drawBtnBg(x, y, _data.keyboard_state.opt, TFT_MAGENTA);
    if (_data.keyboard_state.opt) {
        GetHAL().canvasKeyboardBar.pushImage(x, y, width, height, image_data_opt);
    } else {
        GetHAL().canvasKeyboardBar.pushImage(x, y, width, height, image_data_opt1);
    }

    y = y + height + gap_y;
    drawBtnBg(x, y, _data.keyboard_state.alt, TFT_GREEN);
    if (_data.keyboard_state.alt) {
        GetHAL().canvasKeyboardBar.pushImage(x, y, width, height, image_data_alt);
    } else {
        GetHAL().canvasKeyboardBar.pushImage(x, y, width, height, image_data_alt0);
    }

    // Push
    GetHAL().pushCanvasKeyboardBar();
}

void Launcher::start_keyboard_bar()
{
    GetHAL().keyboard.onKeyEvent.connect([this](const Keyboard::KeyEvent_t& keyEvent) {
        bool need_render = false;
        switch (keyEvent.keyCode) {
            case KEY_CAPSLOCK:
                _data.keyboard_state.caps_lock = keyEvent.state;
                need_render                    = true;
                break;
            case KEY_LEFTCTRL:
                _data.keyboard_state.ctrl = keyEvent.state;
                need_render               = true;
                break;
            case KEY_LEFTMETA:
                _data.keyboard_state.opt = keyEvent.state;
                need_render              = true;
                break;
            case KEY_LEFTALT:
                _data.keyboard_state.alt = keyEvent.state;
                need_render              = true;
                break;
            case KEY_LEFTSHIFT:
                _data.keyboard_state.fn = keyEvent.state;
                need_render             = true;
                break;
            default:
                break;
        }

        if (need_render) {
            render_keyboard_bar();
        }
    });

    render_keyboard_bar();
}
